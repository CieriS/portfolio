CREATE DATABASE EmporioSolidale;

USE EmporioSolidale;

CREATE TABLE IF NOT EXISTS Richiedente(

    Cod_Fiscale CHAR(16) NOT NULL,
    Cognome VARCHAR(30) NOT NULL,
    Nome VARCHAR(30) NOT NULL,
    LuogoNascita VARCHAR(30) NOT NULL,
    DataNascita DATE NOT NULL,
    Cittadinanza VARCHAR(15) NOT NULL,
    InItaliaDallAnno YEAR,
    Residenza VARCHAR(30) NOT NULL,
    Via VARCHAR(30) NOT NULL,
    TipoAlloggio VARCHAR(18) NOT NULL,
    DocumentoIdent CHAR(9) NULL,
    DocumentoImmig CHAR(10) NULL,
    Telefono VARCHAR(15) NOT NULL,
    Mail VARCHAR(50) NOT NULL,
    ServiziSociali VARCHAR(15) NOT NULL,
    ComeHaConosciutoEmporio VARCHAR(30) NULL,
    nMembri INT(2) NOT NULL,
    DataOraRitiroTessera DATE DEFAULT NULL,
    DataOraStalloTessera DATE DEFAULT NULL,
    ISEE INT NOT NULL,
    PRIMARY KEY(Cod_Fiscale)
);


CREATE TABLE IF NOT EXISTS Tessera(

    Codice CHAR(12) NOT NULL,

CREATE TABLE IF NOT EXISTS TesseraRilasciata(

    CodiceTessera CHAR(12) NOT NULL,
    CodFamiglia CHAR(16) NOT NULL,
    DataOraRilascioTessera DATE DEFAULT NULL,
    DataOraScadenzaTessera DATE DEFAULT NULL,
    PRIMARY KEY(CodiceTessera, CodFamiglia),
    FOREIGN KEY(CodiceTessera) REFERENCES Tessera(Codice) ON UPDATE CASCADE ON DELETE CASCADE,
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE IF NOT EXISTS StoricoTessere(

    OrdineCronologico INT AUTO_INCREMENT NOT NULL,
    CodiceTessera CHAR(12) NOT NULL,
    CodFamiglia CHAR(16) NOT NULL,
    DataRilascio DATE NOT NULL,
    DataRitiro DATE DEFAULT NULL,
    PRIMARY KEY( OrdineCronologico, CodiceTessera, CodFamiglia ),
    FOREIGN KEY(CodiceTessera) REFERENCES Tessera(Codice) ON UPDATE CASCADE ON DELETE CASCADE,
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);




CREATE TABLE IF NOT EXISTS Punteggio(

    CodFamiglia CHAR(16) NOT NULL,
    PunteggioISEE INT DEFAULT 0 NULL,
    PunteggioPresenzaMinori INT DEFAULT 0 NULL,
    PunteggioDisoccupazione INT DEFAULT 0 NULL,
    PunteggioInvalidi INT DEFAULT 0 NULL,
    PunteggioSituazioneDebitoria INT DEFAULT 0 NULL,
    PunteggioBenefit INT DEFAULT 0 NULL,
    PunteggioTotale INT DEFAULT 0 NULL,
    PRIMARY KEY(CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS DispCollaborative(

    CodFamiglia CHAR(16) NOT NULL,    
    AmbitoFormativo TINYTEXT NULL,
    AmbitoLavorativo TINYTEXT NULL,
    AmbitoSociale TINYTEXT NULL,
    ValutazioneCommissione TINYTEXT NULL, 
    PRIMARY KEY(CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE IF NOT EXISTS SituazioneSocioSanitaria(

    CodFamiglia CHAR(16) NOT NULL, 
    N INT NOT NULL,
    Evento VARCHAR(60) NOT NULL,
    Anno YEAR NULL,
    DocAll VARCHAR(30) NULL,
    PRIMARY KEY(N, CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE 
);




CREATE TABLE IF NOT EXISTS PatrAtt(

    CodFamiglia CHAR(16) NOT NULL,
    N INT NOT NULL,
    Elemento VARCHAR(60) NOT NULL,
    Valore DECIMAL(8,2) NULL,
    PRIMARY KEY(N, CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE IF NOT EXISTS PatrPass(

    CodFamiglia CHAR(16) NOT NULL,
    N INT NOT NULL,
    Elemento VARCHAR(60) NOT NULL,
    Valore DECIMAL(8,2) NULL,
    Scadenza DATE NULL,
    PRIMARY KEY(N, CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE IF NOT EXISTS Entrate(

    CodFamiglia CHAR(16) NOT NULL,
    N INT NOT NULL,
    Elemento VARCHAR(60) NOT NULL,
    Valore DECIMAL(8,2) NULL,
    Scadenza DATE NULL,
    PRIMARY KEY(N, CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE IF NOT EXISTS Uscite(

    CodFamiglia CHAR(16) NOT NULL,
    N INT NOT NULL,
    Elemento VARCHAR(60) NOT NULL,
    Valore DECIMAL(8,2) NULL,
    Scadenza DATE NULL,
    PRIMARY KEY(N, CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE IF NOT EXISTS Membri(

    nMembro INT(2) NOT NULL,
    NomeCognome VARCHAR(60) NULL,
    LuogoNascita VARCHAR(30) NULL,
    DataNascita DATE NULL,
    Parentela  VARCHAR(15) NOT NULL,
    Occupazione  VARCHAR(20) NULL,
    TitoloDiStudio VARCHAR(20) NULL,
    Competenze VARCHAR(20) NULL,
    ConoscenzaLingua VARCHAR(30) NULL,
    Patente VARCHAR(4) NULL,
    AttualeAtt VARCHAR(15) NULL,
    Presso VARCHAR(15) NULL,
    Termine DATE NULL,
    SenzaLavoroDa DATE NULL,
    CodFamiglia CHAR(16) NOT NULL,
    PRIMARY KEY(nMembro, CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE IF NOT EXISTS Competenze(

    Nome VARCHAR(30) NOT NULL,
    Descrizione TINYTEXT,
    Presso VARCHAR(30),
    PRIMARY KEY ( Nome )
);



CREATE TABLE IF NOT EXISTS SkillsMembri(

    nMembro INT(2) NOT NULL,
    CodFamiglia CHAR(16) NOT NULL,
    Nome VARCHAR(30) NOT NULL,
    PRIMARY KEY ( nMembro, CodFamiglia, Nome ),
    FOREIGN KEY ( nMembro ) REFERENCES Membri(nMembro) ON UPDATE CASCADE ON DELETE CASCADE,
    FOREIGN KEY ( CodFamiglia ) REFERENCES Membri(CodFamiglia) ON UPDATE CASCADE ON DELETE CASCADE,
    FOREIGN KEY ( Nome ) REFERENCES Competenze(Nome) ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE VIEW Pass AS 
SELECT R.Cod_Fiscale, R.Cognome, R.Nome, R.nMembri, P.PunteggioISEE, P.PunteggioPresenzaMinori, P.PunteggioDisoccupazione, P.PunteggioInvalidi, P.PunteggioSituazioneDebitoria, P.PunteggioBenefit, P.PunteggioTotale, TR.CodiceTessera, DATE_FORMAT(TR.DataOraRilascioTessera, '%d/%m/%Y') AS DataOraRilascioTessera, DATE_FORMAT(TR.DataOraScadenzaTessera, '%d/%m/%Y') AS DataOraScadenzaTessera, DATE_FORMAT(R.DataOraRitiroTessera, '%d/%m/%Y') AS DataOraRitiroTessera, DATE_FORMAT(R.DataOraStalloTessera, '%d/%m/%Y') AS DataOraStalloTessera  
FROM Richiedente AS R LEFT OUTER JOIN TesseraRilasciata AS TR 
ON R.Cod_Fiscale = TR.CodFamiglia 
LEFT OUTER JOIN  Tessera AS T 
ON TR.CodiceTessera = T.Codice 
INNER JOIN Punteggio AS P 
ON R.Cod_Fiscale = P.CodFamiglia  
ORDER BY P.PunteggioTotale DESC;
