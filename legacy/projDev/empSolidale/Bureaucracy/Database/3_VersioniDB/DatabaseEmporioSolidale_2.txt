CREATE DATABASE EmporioSolidale;

USE EmporioSolidale;



CREATE TABLE Richiedente(

    Cod_Fiscale CHAR(16) NOT NULL,
    Cognome VARCHAR(30) NOT NULL,
    Nome VARCHAR(30) NOT NULL,
    LuogoNascita VARCHAR(30) NOT NULL,
    DataNascita DATE NOT NULL,
    Cittadinanza VARCHAR(15) NOT NULL,
    InItaliaDallAnno YEAR(4) NULL,
    Residenza VARCHAR(30) NOT NULL,
    Via VARCHAR(30) NOT NULL,
    TipoAlloggio VARCHAR(18) NOT NULL,
    DocumentoIdent CHAR(9) NOT NULL,
    DocumentoImmig CHAR(100) NULL,
    Telefono CHAR(10) NOT NULL,
    Mail VARCHAR(30) NOT NULL,
    ServiziSociali VARCHAR(15) NOT NULL,
    ComeHaConosciutoEmporio VARCHAR(30) NULL,
    TipoDomanda VARCHAR(20) NOT NULL,
    nMembri INT NOT NULL,
    ISEE INT NOT NULL,
    PRIMARY KEY(Cod_Fiscale)
);





CREATE TABLE TesseraRilasciata(

    CodiceTessera CHAR(10) DEFAULT NULL,
    CodFamiglia CHAR(16) NOT NULL,
    DataOraRilascioTessera DATE DEFAULT NULL,
    DataOraNuovaTessera DATE DEFAULT NULL,
    Rilascio CHAR(2) NOT NULL,
    PRIMARY KEY(CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);




CREATE TABLE Punteggio(

    CodFamiglia CHAR(16) NOT NULL,
    PunteggioISEE INT DEFAULT 0 NULL,
    PunteggioPresenzaMinori INT DEFAULT 0 NULL,
    PunteggioDisoccupazione INT DEFAULT 0 NULL,
    PunteggioInvalidi INT DEFAULT 0 NULL,
    PunteggioSituazioneDebitoria INT DEFAULT 0 NULL,
    PunteggioBenefit INT DEFAULT 0 NULL,
    PunteggioTotale INT DEFAULT 0 NULL,
    PRIMARY KEY(CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE TABLE DispCollaborative(

    CodFamiglia CHAR(16) NOT NULL,    
    AmbitoFormativo TINYTEXT NULL,
    AmbitoLavorativo TINYTEXT NULL,
    AmbitoSociale TINYTEXT NULL,
    PRIMARY KEY(CodFamiglia),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE SituazioneSocioSanitaria(

    CodFamiglia CHAR(16) NOT NULL, 
    N INT AUTO_INCREMENT NOT NULL,
    Evento VARCHAR(35) NULL,
    Anno YEAR(4) NULL,
    DocAll VARCHAR(30) NULL,
    PRIMARY KEY(N),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE 
);




CREATE TABLE PatrAtt(

    CodFamiglia CHAR(16) NOT NULL,
    N INT AUTO_INCREMENT NOT NULL,
    Elemento VARCHAR(35) NULL,
    Valore DECIMAL(8,2) NULL,
    PRIMARY KEY(N),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE PatrPass(

    CodFamiglia CHAR(16) NOT NULL,
    N INT AUTO_INCREMENT NOT NULL,
    Elemento VARCHAR(35) NULL,
    Valore DECIMAL(8,2) NULL,
    Scadenza DATE NOT NULL,
    PRIMARY KEY(N),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE Entrate(

    CodFamiglia CHAR(16) NOT NULL,
    N INT AUTO_INCREMENT NOT NULL,
    Elemento VARCHAR(35) NULL,
    Valore DECIMAL(8,2) NULL,
    Scadenza DATE NOT NULL,
    PRIMARY KEY(N),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE Uscite(

    CodFamiglia CHAR(16) NOT NULL,
    N INT AUTO_INCREMENT NOT NULL,
    Elemento VARCHAR(35) NULL,
    Valore DECIMAL(8,2) NULL,
    Scadenza DATE NOT NULL,
    PRIMARY KEY(N),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);



CREATE TABLE Membri(

    nMembro INT AUTO_INCREMENT NOT NULL,
    NomeCognome VARCHAR(60) NOT NULL,
    LuogoNascita VARCHAR(30) NOT NULL,
    DataNascita DATE NOT NULL,
    Parentela  VARCHAR(10) NOT NULL,
    Occupazione  VARCHAR(20) NOT NULL,
    TitoloDiStudio VARCHAR(20) NOT NULL,
    Competenze VARCHAR(50) NOT NULL,
    ConoscenzaLingua VARCHAR(30) NULL,
    Patente VARCHAR(5) NULL,
    AttualeAtt VARCHAR(15) NULL,
    Presso VARCHAR(15) NULL,
    Termine VARCHAR(15) NULL,
    SenzaLavoroDa VARCHAR(15) NULL,
    CodFamiglia CHAR(16) NOT NULL,
    PRIMARY KEY(nMembro),
    FOREIGN KEY(CodFamiglia) REFERENCES Richiedente(Cod_Fiscale) ON UPDATE CASCADE ON DELETE CASCADE
);

    Username VARCHAR(15) NOT NULL,
    Password CHAR(32) NOT NULL,
