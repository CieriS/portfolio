CREATE TABLE Cliente(
    id INT AUTO_INCREMENT NOT NULL,
    mail VARCHAR (40) NOT NULL,
    nome VARCHAR (30) NOT NULL,
    cognome VARCHAR (30) NOT NULL,
    telefono VARCHAR (15) NOT NULL,
    cifraRichiesta VARCHAR (20) NOT NULL,
    date DATETIME NOT NULL,
    PRIMARY KEY(id)
);

CREATE VIEW listaClienti AS 
SELECT id AS "Identificatore", mail AS 'E-Mail', cognome AS 'Cognome', nome AS 'Nome', telefono AS 'Telefono', cifraRichiesta AS 'Quantitativo Richiesto', date AS 'Data Della Richiesta' 
FROM Cliente 
GROUP BY mail, telefono 
ORDER BY cognome, nome ASC;
